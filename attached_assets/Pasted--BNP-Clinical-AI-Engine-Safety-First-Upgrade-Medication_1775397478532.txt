# ================= INTENT DETECTION =================

def detect_intent(q: str):
ql = q.lower()
if any(w in ql for w in ["dose", "جرعة", "mg", "ml"]):
return "drug"
if any(w in ql for w in ["protocol", "sepsis", "icu"]):
return "protocol"
return "general"

# ================= SAFETY ENGINE =================

class SafetyEngine:
@staticmethod
def calculate_dose(drug: str, weight: float):
if drug not in DRUG_DB:
return None, []
d = DRUG_DB[drug]
dose = d["dose_per_kg"] * weight
alerts = []
if dose > d["max_dose"]:
alerts.append("🚨 OVERDOSE: exceeds maximum safe dose")
return dose, alerts

```
@staticmethod
def check_contraindications(drug: str, conditions: List[str]):
    alerts = []
    if drug not in DRUG_DB:
        return alerts
    for c in conditions or []:
        if c.lower() in [x.lower() for x in DRUG_DB[drug]["contraindications"]]:
            alerts.append(f"⚠️ Contraindication: {c}")
    return alerts

@staticmethod
def check_interactions(drug: str, other_drugs: List[str]):
    alerts = []
    if drug not in DRUG_DB:
        return alerts
    for od in other_drugs or []:
        if od.lower() in [x.lower() for x in DRUG_DB[drug]["interactions"]]:
            alerts.append(f"⚠️ Interaction with {od}")
    return alerts

@staticmethod
def high_risk_flag(drug: str):
    if drug in DRUG_DB and DRUG_DB[drug]["high_risk"]:
        return ["🔴 High Risk Medication"]
    return []
```
