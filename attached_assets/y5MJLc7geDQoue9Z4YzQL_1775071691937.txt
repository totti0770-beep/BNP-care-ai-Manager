export const processPDF = async (filePath, fileName, categoryId) => {
const db = await getDBConnection();
try {
// 1. حساب SHA-256 لضمان أصالة المرجع
const fileData = await RNFS.readFile(filePath, 'base64');
const checksum = CryptoJS.SHA256(fileData).toString();
// 2. استخراج النص (ملاحظة: في React Native نستخدم عادة مكتبة Native أو Bridge)
// سنفترض هنا دالة تجريبية لاستخراج النص صفحة بصفحة
const extractedPages = await simulatePdfExtraction(filePath); 
for (let page of extractedPages) {
const chunks = chunkText(page.text, 500); // تقسيم كل صفحة لقطع (500 حرف)
for (let chunk of chunks) {
await db.executeSql(
`INSERT INTO KnowledgeBase (category_id, content, source_name, page_number, checksum) 
VALUES (?, ?, ?, ?, ?)`,
[categoryId, chunk, fileName, page.pageNumber, checksum]
);
}
}
return true;
} catch (error) {
console.error("PDF Processing Error: ", error);
return false;
}
};
// تقسيم النص للحفاظ على السياق (Context Window) للـ AI
const chunkText = (text, size) => {
const chunks = [];
for (let i = 0; i < text.length; i += size) {
chunks.push(text.substring(i, i + size));
}
return chunks;
};

answer: "عذراً، هذه المعلومة غير متوفرة في المراجع المرفوعة حالياً. يرجى الرجوع للمشرف.",

};
